import sys, os, tempfile
sys.path.insert(0, 'src')

from pathlib import Path
from sonya.state.substrate import Substrate

tmp = Path(tempfile.mkdtemp()) / "test_migration.db"
sub = Substrate.open(tmp)
print("Schema version:", sub.schema_version)

from sonya.memory.types import RecordType, Scope, RetentionPolicy
from sonya.memory.trace_layer import TraceLayer
from sonya.memory.procedural import ProceduralMemory
from sonya.memory.semantic import SemanticMemory
from sonya.memory.episodic import EpisodicMemory

trace = TraceLayer(sub)
entry = trace.record(
    record_type=RecordType.tool_observation,
    raw_content="Test trace content",
    normalized_summary="Test trace summary",
    source="test",
    scope=Scope.global_,
    importance=0.5,
)
print("Trace entry:", entry.trace_id, entry.record_type, entry.scope)

proc = ProceduralMemory(sub)
lesson = proc.add_lesson(
    statement="Test lesson: shell.exec works",
    domain="tool_usage",
    pattern="shell",
    confidence=0.7,
    scope=Scope.global_,
)
print("Lesson:", lesson.lesson_id, lesson.domain, lesson.confidence)

semantic = SemanticMemory(sub)
fact = semantic.add_fact(
    fact_type="test_fact",
    statement="Test: Sonya remembers Ivan",
    confidence=0.8,
    scope="global",
    project_id="",
    retention_policy="long",
)
print("Fact:", fact.fact_id, fact.scope, fact.project_id, fact.retention_policy)

episodic = EpisodicMemory(sub)
event = episodic.record(
    event_type="dialogue_event",
    raw_content="Hello from test",
    normalized_summary="Test: user said hello",
    source="user",
    importance_score=0.5,
    record_type="dialogue_event",
    scope="main_chat",
    project_id="",
    retention_policy="medium",
)
print("Event:", event.event_id, event.record_type, event.scope, event.retention_policy)

recent = episodic.get_recent(limit=5, project_id=None, exclude_trace_types=True)
print("Recent events:", len(recent))

from sonya.memory.compiler import MemoryCompiler
compiler = MemoryCompiler(sub)
results = compiler.run(since_hours=24)
print("Compiler results:", results)

cur = sub.connection.execute("SELECT name FROM sqlite_master WHERE type='table'")
tables = [r[0] for r in cur.fetchall()]
print("raw_traces:", "raw_traces" in tables)
print("procedural_memory:", "procedural_memory" in tables)

sub.close()
os.unlink(tmp)
print("ALL TESTS PASSED")
